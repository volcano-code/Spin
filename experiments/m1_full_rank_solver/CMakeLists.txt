cmake_minimum_required(VERSION 3.16)
project(FullRankHollowingContract LANGUAGES CXX)
include(CTest)
get_filename_component(FULL_RANK_SOLVER_ROOT_DEFAULT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FULL_RANK_SOLVER_ROOT "${FULL_RANK_SOLVER_ROOT_DEFAULT}" CACHE PATH "Isolated adapter source")
set(MOTION_ROOT "${FULL_RANK_SOLVER_ROOT}" CACHE PATH "Differential source snapshot")
set(MIS_NATIVE_ROOT "${MOTION_ROOT}" CACHE PATH "Native source snapshot")
set(FROZEN_GENERAL_BUILD "" CACHE PATH "Optional read-only validated general build")
if(FROZEN_GENERAL_BUILD)
  set(MIS_EIGEN_INCLUDE "" CACHE PATH "Eigen headers matching frozen core")
  find_library(MIS_GMP_LIBRARY gmp REQUIRED)
  find_library(MIS_MPFR_LIBRARY mpfr REQUIRED)
  find_library(MIS_BOOST_THREAD_LIBRARY boost_thread REQUIRED)
  find_library(MIS_BOOST_SYSTEM_LIBRARY boost_system REQUIRED)
  add_library(spin_native_frozen STATIC IMPORTED)
  set_target_properties(spin_native_frozen PROPERTIES IMPORTED_LOCATION "${FROZEN_GENERAL_BUILD}/native/libmake_it_stand_core.a"
    INTERFACE_INCLUDE_DIRECTORIES "${MIS_NATIVE_ROOT}/include;${MIS_EIGEN_INCLUDE}"
    INTERFACE_LINK_LIBRARIES "${MIS_GMP_LIBRARY};${MIS_MPFR_LIBRARY};${MIS_BOOST_THREAD_LIBRARY};${MIS_BOOST_SYSTEM_LIBRARY}")
  add_library(spin_general_differential STATIC IMPORTED)
  set_target_properties(spin_general_differential PROPERTIES IMPORTED_LOCATION "${FROZEN_GENERAL_BUILD}/libspin_general_differential.a"
    INTERFACE_INCLUDE_DIRECTORIES "${MIS_NATIVE_ROOT}/include;${MOTION_ROOT}/include;${MIS_EIGEN_INCLUDE}"
    INTERFACE_LINK_LIBRARIES spin_native_frozen)
  foreach(name IN ITEMS polyhedral_moment_differential truncation_root_certificate quadric_source_motion full_rank_moment)
    add_test(NAME general-${name} COMMAND "${FROZEN_GENERAL_BUILD}/${name}_regression")
  endforeach()
else()
  add_subdirectory("${MOTION_ROOT}/experiments/m1_general_differential" general)
endif()
add_library(spin_full_rank_solver STATIC
  "${FULL_RANK_SOLVER_ROOT}/src/spin/FullRankHollowingSolver.cpp"
  "${FULL_RANK_SOLVER_ROOT}/src/spin/FasterTheoremConditions.cpp")
target_include_directories(spin_full_rank_solver BEFORE PUBLIC "${MIS_NATIVE_ROOT}/include" "${MOTION_ROOT}/include" "${FULL_RANK_SOLVER_ROOT}/include")
target_compile_features(spin_full_rank_solver PUBLIC cxx_std_17)
target_link_libraries(spin_full_rank_solver PUBLIC spin_general_differential)
add_executable(full_rank_hollowing_regression "${FULL_RANK_SOLVER_ROOT}/tests/full_rank_hollowing_regression.cpp")
target_link_libraries(full_rank_hollowing_regression PRIVATE spin_full_rank_solver)
foreach(test IN ITEMS adapter chain negative solver integration modes offtarget-yoyo audit replay)
  add_test(NAME full-rank-${test} COMMAND full_rank_hollowing_regression ${test})
endforeach()
set_tests_properties(full-rank-replay PROPERTIES LABELS "acceptance;reproducibility")
