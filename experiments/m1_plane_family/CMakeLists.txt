cmake_minimum_required(VERSION 3.16)
project(SpinPlaneFamilyContract LANGUAGES CXX)
include(CTest)
get_filename_component(MIS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${MIS_ROOT}" native EXCLUDE_FROM_ALL)
if(NOT MAKE_IT_STAND_BUILD_SPIN_IT_FASTER)
  message(FATAL_ERROR "Plane-family evaluation requires the native Faster evaluator")
endif()
add_library(spin_plane_family STATIC "${MIS_ROOT}/src/spin/PlaneMomentEvaluator.cpp")
target_link_libraries(spin_plane_family PUBLIC make_it_stand_core)
target_compile_features(spin_plane_family PUBLIC cxx_std_17)
add_executable(plane-family-regression "${MIS_ROOT}/tests/plane_family_regression.cpp")
target_link_libraries(plane-family-regression PRIVATE spin_plane_family)
add_test(NAME plane-family-contract COMMAND plane-family-regression)
